import { expect, test, type Page } from "@playwright/test";
import path from "node:path";

const organizationSlug = "minh-khai-university";
const password = "Demo123!";
const screenshotDir = path.resolve("artifacts/ui-review");
const uploadPath = path.resolve("artifacts/ui-review/overview-vi-mobile.png");

async function assertNoHorizontalOverflow(page: Page) {
  const dimensions = await page.evaluate(() => ({
    viewport: document.documentElement.clientWidth,
    content: document.documentElement.scrollWidth
  }));
  expect(dimensions.content).toBeLessThanOrEqual(dimensions.viewport + 1);
}

async function login(page: Page, email: string) {
  await page.getByLabel("Mã tổ chức").fill(organizationSlug);
  await page.getByLabel("Email tổ chức").fill(email);
  await page.getByLabel("Mật khẩu", { exact: true }).fill(password);
  await page.getByRole("button", { name: "Đăng nhập", exact: true }).click();
  await expect(page).toHaveURL(/\/overview$/);
}

test("student intake to human-reviewed assignment", async ({ page }) => {
  const browserErrors: string[] = [];
  page.on("pageerror", (error) => browserErrors.push(error.message));
  page.on("console", (message) => {
    if (message.type() === "error") browserErrors.push(message.text());
  });

  await page.goto("/login");
  await expect(page.getByRole("heading", { name: "Chào mừng trở lại" })).toBeVisible();
  await assertNoHorizontalOverflow(page);
  await page.screenshot({ path: path.join(screenshotDir, "current-login-desktop.png"), fullPage: true });

  await login(page, "student@caseflow.local");
  await expect(page.getByText("Trường Đại học Minh Khai", { exact: true }).first()).toBeVisible();
  await assertNoHorizontalOverflow(page);
  await page.screenshot({ path: path.join(screenshotDir, "current-overview-desktop.png"), fullPage: true });

  await page.goto("/cases/new");
  const unique = Date.now().toString().slice(-6);
  await page.getByLabel("Tiêu đề").fill(`[E2E] Không đăng nhập được SIS ${unique}`);
  await page.getByLabel("Mô tả chi tiết").fill(
    "Em không đăng nhập được tài khoản SIS dù đã thử đặt lại mật khẩu hai lần và cần hỗ trợ trước ngày mai."
  );
  await page.locator('input[type="file"]').setInputFiles(uploadPath);
  await page.getByRole("button", { name: "Phân tích bằng AI" }).click();
  await expect(page.locator(".analysis-content")).toBeVisible();
  await expect(page.getByText("Trung tâm CNTT", { exact: true })).toBeVisible();
  await page.screenshot({ path: path.join(screenshotDir, "current-intake-analysis.png"), fullPage: true });

  await page.getByRole("button", { name: "Tạo hồ sơ" }).click();
  await expect(page).toHaveURL(/\/cases\/[^/]+$/);
  const caseCode = (await page.locator(".case-code").textContent())?.trim() || "";
  expect(caseCode).toMatch(/^CF-/);
  await expect(page.getByText("overview-vi-mobile.png", { exact: true })).toBeVisible();
  await expect(page.getByText("Đang chờ xác nhận", { exact: true })).toBeVisible();
  await page.screenshot({ path: path.join(screenshotDir, "current-case-student.png"), fullPage: true });

  await page.getByRole("button", { name: "Đăng xuất" }).click();
  await expect(page).toHaveURL(/\/login$/);
  await login(page, "manager@caseflow.local");

  await page.getByRole("button", { name: "Thông báo" }).click();
  const caseNotification = page.locator(".notification-item").filter({ hasText: caseCode }).first();
  await expect(caseNotification).toBeVisible();
  await page.screenshot({ path: path.join(screenshotDir, "current-notifications.png"), fullPage: true });
  await caseNotification.click();
  await expect(page.locator(".case-code")).toHaveText(caseCode);

  await page.getByRole("button", { name: "Xác nhận phân luồng" }).click();
  await expect(page.getByText("Đã xác nhận", { exact: true })).toBeVisible();
  await page.getByLabel("Người xử lý").selectOption({ label: "Lê Hoàng Nam · Trung tâm CNTT" });
  await expect(page.getByLabel("Người xử lý")).toHaveValue(/.+/);

  const downloadPromise = page.waitForEvent("download");
  await page.getByRole("button", { name: /Tải tệp: overview-vi-mobile\.png/ }).click();
  const download = await downloadPromise;
  expect(download.suggestedFilename()).toBe("overview-vi-mobile.png");
  await assertNoHorizontalOverflow(page);
  await page.screenshot({ path: path.join(screenshotDir, "current-case-manager.png"), fullPage: true });

  await page.setViewportSize({ width: 390, height: 844 });
  await assertNoHorizontalOverflow(page);
  await page.getByRole("button", { name: "Thông báo" }).click();
  await expect(page.locator(".notification-popover")).toBeVisible();
  await page.screenshot({ path: path.join(screenshotDir, "current-case-mobile-notifications.png"), fullPage: true });
  await page.getByRole("button", { name: "Thông báo" }).click();

  await page.getByRole("button", { name: "EN", exact: true }).click();
  await expect(page.getByText("Case detail", { exact: true })).toBeVisible();
  await assertNoHorizontalOverflow(page);
  await page.screenshot({ path: path.join(screenshotDir, "current-case-mobile-en.png"), fullPage: true });

  expect(browserErrors).toEqual([]);
});
